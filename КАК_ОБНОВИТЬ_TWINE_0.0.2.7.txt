window.advanceTime = function(minutes, ignoreNeeds) {
  var game = State.variables.game;
  if (!game || !game.time) return;
  var t = game.time;
  t.minute += minutes;
  while (t.minute >= 60) {
    t.minute -= 60;
    t.hour += 1;
    if (t.hour >= 24) {
      t.hour = 0;
      t.totalDays += 1;
      t.weekday = getWeekday(t.totalDays);
      var daysInMonth = [31,28,31,30,31,30,31,31,30,31,30,31];
      var days = t.totalDays;
      var year = 1;
      while (days > 365) {
        days -= 365;
        year++;
      }
      var month = 1;
      while (days > daysInMonth[month-1]) {
        days -= daysInMonth[month-1];
        month++;
      }
      t.day = days;
      t.month = month;
      t.year = year;
    }
  }
  updateNeeds(minutes, ignoreNeeds);
};

window.updateNeeds = function(minutes, ignoreNeeds) {
  var p = State.variables.game.player;
  if (!p || !p.needs) return;
  if (!ignoreNeeds) ignoreNeeds = [];
  if (!Array.isArray(ignoreNeeds)) ignoreNeeds = [ignoreNeeds];
  function isIgnored(need) {
    return ignoreNeeds.indexOf(need) !== -1;
  }
  var hours = minutes / 60;
  if (p.sleeping) {
    if (!isIgnored('energy')) p.needs.energy = Math.min(100, p.needs.energy + 15 * hours);
    if (!isIgnored('satiety')) p.needs.satiety = Math.max(0, p.needs.satiety - 4 * hours);
    if (!isIgnored('hygiene')) p.needs.hygiene = Math.max(0, p.needs.hygiene - 2 * hours);
    if (!isIgnored('bladder')) p.needs.bladder = Math.max(0, p.needs.bladder - 6 * hours);
  } else {
    if (!isIgnored('energy')) p.needs.energy = Math.max(0, p.needs.energy - 4 * hours);
    if (!isIgnored('satiety')) p.needs.satiety = Math.max(0, p.needs.satiety - 5 * hours);
    if (!isIgnored('hygiene')) p.needs.hygiene = Math.max(0, p.needs.hygiene - 2 * hours);
    if (!isIgnored('social')) p.needs.social = Math.max(0, p.needs.social - 2 * hours);
    if (!isIgnored('fun')) p.needs.fun = Math.max(0, p.needs.fun - 2 * hours);
    if (!isIgnored('bladder')) p.needs.bladder = Math.max(0, p.needs.bladder - 4 * hours);
  }
  p.needs.energy = Math.min(100, Math.max(0, p.needs.energy));
  p.needs.satiety = Math.min(100, Math.max(0, p.needs.satiety));
  p.needs.hygiene = Math.min(100, Math.max(0, p.needs.hygiene));
  p.needs.social = Math.min(100, Math.max(0, p.needs.social));
  p.needs.fun = Math.min(100, Math.max(0, p.needs.fun));
  p.needs.bladder = Math.min(100, Math.max(0, p.needs.bladder));
};

window.getWeekday = function(totalDays) {
  var weekdays = ["Понедельник", "Вторник", "Среда", "Четверг", "Пятница", "Суббота", "Воскресенье"];
  return weekdays[(totalDays - 1) % 7];
};

window.formatTime = function() {
  var game = State.variables.game;
  if (!game || !game.time) return "--:--";
  var t = game.time;
  var h = t.hour;
  var m = t.minute;
  return h + ':' + (m < 10 ? '0' : '') + m;
};

window.checkCriticalNeeds = function() {
  var p = State.variables.game.player;
  if (!p || !p.needs) return;
  if (p.needs.bladder <= 0) {
    p.needs.bladder = 100;
    p.needs.hygiene = Math.max(0, p.needs.hygiene - 30);
  }
};

window.updateMood = function() {
  var p = State.variables.game.player;
  if (!p || !p.needs) return;
  var n = p.needs;
  var avg = (n.energy + n.satiety + n.hygiene + n.social + n.fun + n.bladder) / 6;
  var penalty = 0;
  if (n.energy < 30) penalty += 8;
  if (n.satiety < 30) penalty += 10;
  if (n.hygiene < 30) penalty += 5;
  if (n.social < 30) penalty += 5;
  if (n.fun < 30) penalty += 5;
  if (n.bladder < 30) penalty += 10;
  if (n.energy < 10) penalty += 10;
  if (n.satiety < 10) penalty += 10;
  if (n.bladder < 10) penalty += 10;
  n.mood = Math.round(Math.min(100, Math.max(0, avg - penalty)));
};

window.setupFadeTransition = function() {
  var overlay = document.getElementById("screen-fade");
  if (!overlay) {
    overlay = document.createElement("div");
    overlay.id = "screen-fade";
    document.body.appendChild(overlay);
  }
  window.__fadeTimer = null;

  var darken = function() {
    if (window.__fadeTimer) { clearTimeout(window.__fadeTimer); window.__fadeTimer = null; }
    overlay.classList.add("fading");
  };
  var lighten = function() {
    overlay.classList.remove("fading");
  };

  // Затемнение при клике по переходу/действию/смене комнаты в игровой области
  jQuery(document).on("click",
    "#passages a.link-internal.macro-link, #passages button.macro-button",
    function() {
      darken();
      // страховка: если переход не состоялся (открыт alert и т.п.) — вернуть яркость
      window.__fadeTimer = setTimeout(lighten, 500);
    }
  );

  // Осветление после того как новый пассаж показан
  jQuery(document).on(":passagedisplay", function() {
    if (window.__fadeTimer) { clearTimeout(window.__fadeTimer); window.__fadeTimer = null; }
    setTimeout(lighten, 60);
  });
};
jQuery(window).on("load", function() { setTimeout(window.setupFadeTransition, 50); });
jQuery(function() { try { window.setupFadeTransition(); } catch (e) {} });

1. Сохраните историю (Twine сохраняет автоматически, но можно нажать Ctrl+S).
2. Нажмите "Играть"/"Тестировать" (Play/Test) — откроется вкладка с игрой.
3. Проверьте:
   - сайдбар слева, в рамке "Потребности": 6 шкал в 2 столбца,
     шкала "Настроение" широкой полосой внизу;
   - шкалы цветные (на старте зелёные); чит "на 0" -> красные, "на 100" -> зелёные;
   - при клике по действию/переходу экран плавно затемняется и осветляется.

ЕСЛИ ЧТО-ТО НЕ ТАК:
- Шкалы серые/нет цвета -> проверьте, что вставили БЛОК 2 (JavaScript) ЦЕЛИКОМ,
  и что в StoryCaption стоит строка  <<print window.renderNeedsPanel()>>.
- Шкалы в столбик -> проверьте, что пассаж StoryCaption начинается с <<nobr>>
  и заканчивается <</nobr>> (это уже есть в блоке 7).
- Нет затемнения -> проверьте БЛОК 1 (там есть #screen-fade) и БЛОК 2
  (там есть setupFadeTransition).
- Twine показывает ошибку в красном окне -> значит в JS или в пассаже потерялся
  кусок при копировании; вставьте блок заново целиком.
