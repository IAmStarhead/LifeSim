виджеты SugarCube <<listbox>>. Раньше использовались самодельные <select>, и
Twine перерисовывал пассаж прямо в момент выбора — значение сбрасывалось.

обновляется сразу. Причёски по-прежнему зависят от длины; для «Лысый»
цвет/тип/причёска скрыты. Достаточно заменить пассаж CharacterStep2
(Часть 1) и Story JavaScript (Часть 2).

растительность на лице, волосы на теле, форма носа/глаз/бровей/ушей,
тумблеры веснушки/родинки/шрамы/тату/пирсинг), черты отсортированы и переименованы,
жильё с описанием и реально разными комнатами, dev-жильё «Богатый дом»,

window.advanceTime = function(minutes, ignoreNeeds) {
  var game = State.variables.game;
  if (!game || !game.time) return;
  var t = game.time;
  t.minute += minutes;
  while (t.minute >= 60) {
    t.minute -= 60;
    t.hour += 1;
    if (t.hour >= 24) {
      t.hour = 0;
      t.totalDays += 1;
      t.weekday = getWeekday(t.totalDays);
      var daysInMonth = [31,28,31,30,31,30,31,31,30,31,30,31];
      var days = t.totalDays;
      var year = 1;
      while (days > 365) {
        days -= 365;
        year++;
      }
      var month = 1;
      while (days > daysInMonth[month-1]) {
        days -= daysInMonth[month-1];
        month++;
      }
      t.day = days;
      t.month = month;
      t.year = year;
    }
  }
  updateNeeds(minutes, ignoreNeeds);
};

window.updateNeeds = function(minutes, ignoreNeeds) {
  var p = State.variables.game.player;
  if (!p || !p.needs) return;
  if (!ignoreNeeds) ignoreNeeds = [];
  if (!Array.isArray(ignoreNeeds)) ignoreNeeds = [ignoreNeeds];
  function isIgnored(need) {
    return ignoreNeeds.indexOf(need) !== -1;
  }
  var hours = minutes / 60;
  if (p.sleeping) {
    if (!isIgnored('energy')) p.needs.energy = Math.min(100, p.needs.energy + 15 * hours);
    if (!isIgnored('satiety')) p.needs.satiety = Math.max(0, p.needs.satiety - 4 * hours);
    if (!isIgnored('hygiene')) p.needs.hygiene = Math.max(0, p.needs.hygiene - 2 * hours);
    if (!isIgnored('bladder')) p.needs.bladder = Math.max(0, p.needs.bladder - 6 * hours);
  } else {
    if (!isIgnored('energy')) p.needs.energy = Math.max(0, p.needs.energy - 4 * hours);
    if (!isIgnored('satiety')) p.needs.satiety = Math.max(0, p.needs.satiety - 5 * hours);
    if (!isIgnored('hygiene')) p.needs.hygiene = Math.max(0, p.needs.hygiene - 2 * hours);
    if (!isIgnored('social')) p.needs.social = Math.max(0, p.needs.social - 2 * hours);
    if (!isIgnored('fun')) p.needs.fun = Math.max(0, p.needs.fun - 2 * hours);
    if (!isIgnored('bladder')) p.needs.bladder = Math.max(0, p.needs.bladder - 4 * hours);
  }
  p.needs.energy = Math.min(100, Math.max(0, p.needs.energy));
  p.needs.satiety = Math.min(100, Math.max(0, p.needs.satiety));
  p.needs.hygiene = Math.min(100, Math.max(0, p.needs.hygiene));
  p.needs.social = Math.min(100, Math.max(0, p.needs.social));
  p.needs.fun = Math.min(100, Math.max(0, p.needs.fun));
  p.needs.bladder = Math.min(100, Math.max(0, p.needs.bladder));
};

window.getWeekday = function(totalDays) {
  var weekdays = ["Понедельник", "Вторник", "Среда", "Четверг", "Пятница", "Суббота", "Воскресенье"];
  return weekdays[(totalDays - 1) % 7];
};

window.formatTime = function() {
  var game = State.variables.game;
  if (!game || !game.time) return "--:--";
  var t = game.time;
  var h = t.hour;
  var m = t.minute;
  return h + ':' + (m < 10 ? '0' : '') + m;
};

window.checkCriticalNeeds = function() {
  var p = State.variables.game.player;
  if (!p || !p.needs) return;
  if (p.needs.bladder <= 0) {
    p.needs.bladder = 100;
    p.needs.hygiene = Math.max(0, p.needs.hygiene - 30);
  }
};

window.updateMood = function() {
  var p = State.variables.game.player;
  if (!p || !p.needs) return;
  var n = p.needs;
  var avg = (n.energy + n.satiety + n.hygiene + n.social + n.fun + n.bladder) / 6;
  var penalty = 0;
  if (n.energy < 30) penalty += 8;
  if (n.satiety < 30) penalty += 10;
  if (n.hygiene < 30) penalty += 5;
  if (n.social < 30) penalty += 5;
  if (n.fun < 30) penalty += 5;
  if (n.bladder < 30) penalty += 10;
  if (n.energy < 10) penalty += 10;
  if (n.satiety < 10) penalty += 10;
  if (n.bladder < 10) penalty += 10;
  n.mood = Math.round(Math.min(100, Math.max(0, avg - penalty)));
};

window.getNeedLevel = function(value) {
  if (value < 15) return "critical";
  if (value < 30) return "low";
  if (value < 45) return "below-average";
  if (value < 60) return "average";
  if (value < 75) return "above-average";
  if (value < 90) return "high";
  return "excellent";
};

window.setupFadeTransition = function() {
  var overlay = document.getElementById("screen-fade");
  if (!overlay) {
    overlay = document.createElement("div");
    overlay.id = "screen-fade";
    document.body.appendChild(overlay);
  }
  window.__fadeTimer = null;

  var darken = function() {
    if (window.__fadeTimer) { clearTimeout(window.__fadeTimer); window.__fadeTimer = null; }
    overlay.classList.add("fading");
  };
  var lighten = function() {
    overlay.classList.remove("fading");
  };

  // Затемнение при клике по переходу/действию/смене комнаты в игровой области
  jQuery(document).on("click",
    "#passages a.link-internal.macro-link, #passages button.macro-button",
    function() {
      darken();
      // страховка: если переход не состоялся (открыт alert и т.п.) — вернуть яркость
      window.__fadeTimer = setTimeout(lighten, 500);
    }
  );

  // Осветление после того как новый пассаж показан
  jQuery(document).on(":passagedisplay", function() {
    if (window.__fadeTimer) { clearTimeout(window.__fadeTimer); window.__fadeTimer = null; }
    setTimeout(lighten, 60);
  });
};
jQuery(window).on("load", function() { setTimeout(window.setupFadeTransition, 50); });
jQuery(function() { try { window.setupFadeTransition(); } catch (e) {} });

window.getNeedColor = function(value) {
  value = Number(value) || 0;
  if (value < 10) return "#e05252";       // красный
  else if (value < 25) return "#e58b3c";  // оранжевый
  else if (value < 40) return "#d9c84a";  // жёлтый

window.renderNeedBar = function(icon, name, value, extraClass) {
  value = Number(value) || 0;
  var level = window.getNeedLevel(value);
  var color = window.getNeedColor(value);   // инлайн-цвет — не зависит от CSS темы
  var cls = "need-item" + (extraClass ? " " + extraClass : "");
  return ""
    + "<div class=\"" + cls + "\">"
    +   "<div class=\"need-header\">"
    +     "<span class=\"need-icon\">" + icon + "</span>"
    +     "<span class=\"need-name\">" + name + "</span>"
    +     "<span class=\"need-value\">" + value.toFixed(1) + "</span>"
    +   "</div>"
    +   "<div class=\"need-bar\">"
    +     "<div class=\"need-fill need-" + level + "\" style=\"width:" + value + "%;background-color:" + color + ";\"></div>"
    +   "</div>"
    + "</div>";
};

window.getTraitPoints = function() {
  var t = State.variables.game.player.traits || [];
  var spent = 0;
  window.TRAITS.forEach(function(tr) {
    if (t.indexOf(tr.id) !== -1) spent += (tr.sign === "pos" ? tr.cost : -tr.cost);
  });
  return { budget: window.TRAIT_BUDGET, spent: spent, left: window.TRAIT_BUDGET - spent };
};
window.renderTraits = function() {
  var g = State.variables.game;
  var t = g.player.traits || [];

  function traitBtn(tr) {
    var on = t.indexOf(tr.id) !== -1;
    var sign = tr.sign === "pos" ? ("−" + tr.cost) : ("+" + tr.cost);
    var label = (on ? "✓ " : "") + tr.name + " (" + sign + ")";
    return "<div class=\"cc-trait\"><div class=\"cc-trait-head\">"
      + "<button onclick=\"window.toggleTrait('" + tr.id + "')\">" + label + "</button></div>"
      + "<div class=\"cc-trait-desc\">" + tr.desc + "</div></div>";
  }
  var pos = window.TRAITS.filter(function(x){return x.sign==="pos";}).sort(function(a,b){return a.cost-b.cost;});
  var neg = window.TRAITS.filter(function(x){return x.sign==="neg";}).sort(function(a,b){return a.cost-b.cost;});

window.toggleTrait = function(id) {
  var p = State.variables.game.player;
  if (!Array.isArray(p.traits)) p.traits = [];
  var i = p.traits.indexOf(id);
  if (i !== -1) { p.traits.splice(i, 1); }
  else {
    var tr = null;
    window.TRAITS.forEach(function(x){ if (x.id === id) tr = x; });
    if (!tr) return;
    if (tr.sign === "pos" && window.getTraitPoints().left < tr.cost) {
      window.ccNotify("Не хватает очков черт: возьмите отрицательную черту или уберите другую положительную.");
      return;
    }
    p.traits.push(id);
  }
  window.ccRefresh("CharacterStep3");
};

// Делегирование: один раз на документ (надёжнее inline onchange)
window.initCCControls = function() {
  if (window.__ccControlsInit) return;
  window.__ccControlsInit = true;
  // Нативные списки SugarCube (рост, волосы, борода и т.п.) сами сохраняют значение;
  // после их изменения перерисовываем текущий пассаж редактора, чтобы обновить описание
  // и зависимые списки (длина волос -> доступные причёски).
  jQuery(document).on("change", "#passages .character-creation select.macro-listbox", function() {
    var p = State.passage || "";

    // если она не подходит новой длине — ставим вариант по умолчанию (для лысого — none).
    if (p === "CharacterStep2") {
      var row = jQuery(this).closest(".cc-select-row");
      var label = jQuery.trim(row.find(".cc-slider-label").text());
      if (row.length && label === "Длина:") {
        var a = State.variables.game.player.appearance;

        }
      }

      setTimeout(function(){ Engine.play(State.passage); }, 0);
    }
  });
  // Кнопки-тумблеры особых примет и черт (рендерятся через JS)
  jQuery(document).on("click", "button[data-cc-toggle]", function() {
    window.toggleFeature(this.getAttribute("data-cc-toggle"));

  if (!t.length) return "нет";
  return t.map(window.traitName).join(", ");
};

window.finishCharacterCreation = function() {
  var g = State.variables.game;
  var p = g.player;
  if (!p.name || !String(p.name).trim()) p.name = (p.gender === "жен") ? "Ева" : "Адам";
  var h = window.getHousing(p.housing) || window.HOUSING_OPTIONS[1];
  p.housing = h.id;
  p.money = Math.max(0, (Number(p.money) || 0) - h.rent);  // списываем первый месяц аренды
  g.location = "home";
  g.room = h.startRoom || "living_room";
  g.started = true;                                          // показываем игровой интерфейс
  return h;
};
